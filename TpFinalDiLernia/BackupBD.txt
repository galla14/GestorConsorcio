CREATE   PROCEDURE [dbo].[sp_ConsultarUsuarioIniciarSesion]
    @NombreUsuario NVARCHAR(50),
    @Password      NVARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @PassReal   NVARCHAR(12),
            @Intentos   INT,
            @Activo     BIT,
            @Bloqueado  BIT,
            @RolUsuario NVARCHAR(50);

    BEGIN TRY
        SELECT @PassReal   = Password,
               @Intentos   = IntentosFallidos,
               @Activo     = Activo,
               @Bloqueado  = Bloqueado,
               @RolUsuario = Rol
        FROM   dbo.Usuario
        WHERE  NombreUsuario = @NombreUsuario;

        
        IF @PassReal IS NULL OR @Activo = 0
            RETURN -1;

        
        IF @Bloqueado = 1
            RETURN -2;

        
        IF @PassReal <> @Password COLLATE Latin1_General_CS_AS
        BEGIN
            
            IF @RolUsuario = 'superadmin'
                RETURN -1;

            SET @Intentos = @Intentos + 1;

            
            IF @Intentos >= 3
            BEGIN
                UPDATE dbo.Usuario
                SET    IntentosFallidos = @Intentos, Bloqueado = 1
                WHERE  NombreUsuario = @NombreUsuario;

                RETURN -3;
            END

            
            UPDATE dbo.Usuario
            SET    IntentosFallidos = @Intentos
            WHERE  NombreUsuario = @NombreUsuario;

            RETURN -1;
        END

        
        UPDATE dbo.Usuario
        SET    IntentosFallidos = 0
        WHERE  NombreUsuario = @NombreUsuario;

        IF @RolUsuario = 'superadmin'
            RETURN 1;

        RETURN 2;
    END TRY
    BEGIN CATCH
        THROW;
    END CATCH
END
GO

create   PROCEDURE [dbo].[sp_ModificarNombreUsuario]
	@Id_Usuario uniqueidentifier,
	@NombreUsuario NVARCHAR(12)
	
AS
BEGIN
	SET NOCOUNT ON;
	BEGIN TRY
		
		IF NOT EXISTS (SELECT 1 FROM dbo.Usuario WHERE IdUsuario = @Id_Usuario)
			RETURN -1;

		
		IF EXISTS (SELECT 1 FROM dbo.Usuario
		           WHERE NombreUsuario = @NombreUsuario AND IdUsuario <> @Id_Usuario)
			RETURN -2;

		
		IF LEN(@NombreUsuario) NOT BETWEEN 8 AND 12
			RETURN -3;

		UPDATE dbo.Usuario
		SET    NombreUsuario = @NombreUsuario
		WHERE  IdUsuario = @Id_Usuario;

		RETURN 1;
	END TRY
	BEGIN CATCH
		THROW;
	END CATCH

END
